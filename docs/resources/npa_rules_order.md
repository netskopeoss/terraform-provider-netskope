# netskope_npa_rules_order (Resource)

Manages the list position of NPA policy rules.

Takes a list of rule IDs and reorders them so that the first ID in the list appears first (top of the list), and each subsequent ID appears after the previous one.

Rules should be created separately with `netskope_npa_rules` using `rule_order = { order = "bottom" }` (safe default), then pass their IDs to this resource to set the final list position.

## Example Usage

```terraform
# Create rules at bottom (parallel, safe)
resource "netskope_npa_rules" "deny_blocked" {
  rule_name = "deny-blocked-users"
  enabled   = "1"
  group_id  = data.netskope_npa_policy_groups_list.all.data[0].id

  rule_data = {
    policy_type           = "private-app"
    match_criteria_action = { action_name = "block" }
    private_apps          = [netskope_npa_private_app.app.private_app_name]
    access_method         = ["Client"]
  }

  rule_order = { order = "bottom" }

  lifecycle {
    ignore_changes = [rule_order]
  }
}

resource "netskope_npa_rules" "allow_admins" {
  rule_name = "allow-admin-access"
  enabled   = "1"
  group_id  = data.netskope_npa_policy_groups_list.all.data[0].id

  rule_data = {
    policy_type           = "private-app"
    match_criteria_action = { action_name = "allow" }
    private_apps          = [netskope_npa_private_app.app.private_app_name]
    user_groups           = ["IT-Administrators"]
    access_method         = ["Client"]
  }

  rule_order = { order = "bottom" }

  lifecycle {
    ignore_changes = [rule_order]
  }
}

# Set list position: deny first, then allow
resource "netskope_npa_rules_order" "main" {
  rule_ids = [
    netskope_npa_rules.deny_blocked.id,
    netskope_npa_rules.allow_admins.id,
  ]
}
```

### YAML-Driven Example

```terraform
locals {
  policies   = yamldecode(file("policies.yaml"))
  policy_map = { for p in local.policies : p.name => p }
}

resource "netskope_npa_rules" "bulk" {
  for_each = local.policy_map

  rule_name = each.value.name
  enabled   = "1"
  group_id  = var.group_id

  rule_data = {
    policy_type           = "private-app"
    match_criteria_action = { action_name = each.value.action }
    private_apps          = each.value.apps
    user_groups           = each.value.groups
    access_method         = ["Client"]
  }

  rule_order = { order = "bottom" }

  lifecycle {
    ignore_changes = [rule_order]
  }
}

resource "netskope_npa_rules_order" "main" {
  rule_ids = [for p in local.policies : netskope_npa_rules.bulk[p.name].id]
}
```
